// POWER SERIES OPERATIONS
constexpr int mod = 998244353; // = 2^k * c + 1 | primitív gyöknek jónak kell lennie | 2013265921, 167772161, 2113929217
constexpr int N = 1 << 20; // 2^l, l <= k | max N amit transzformálni lehet

constexpr inline long long mod_pow(long long n, long long h) {
    long long res = 1;
    for(;h;h >>= 1) { if(h&(1LL)) res = res * n % mod; n = n * n % mod; }
    return res;
}

constexpr long long mod_primitive_root(){ // kiszámítja a moduló egy primitív gyökét
    long long primes[64] = {}; int size = 0; long long p = 2, m = mod-1;
    while(p*p <= m) { if(m % p == 0) primes[size++] = p; while(m % p == 0) m /= p; ++p; } if(m > 1) primes[size++] = m;
    for(long long i = 2; i < mod; i++) { bool ok = true; for(int j = 0; j < size; j++) ok = ok && mod_pow(i, (mod - 1) / primes[j]) != 1; if(ok) return i; }
    return -1;
}

constexpr inline long long mod_sqrt(long long a) {
    if (mod_pow(a, mod >> 1) != 1) return 0;
    int Q = (mod - 1) >> (__countr_zero(mod-1));
    long long x = mod_pow(a, (Q + 1) >> 1), y = mod_pow(x, Q);
    for (int k = __countr_zero(mod - 1) - 1; k >= 0; --k)
        if (mod_pow(y, 1 << k) != 1) {
            x *= pow(mod_primitive_root(), mod >> (k + 2));
            y *= pow(mod_primitive_root(), mod >> (k + 1));
        }
    return min(x, mod - x);
}

long long w[N];
long long invi[N + 1];
__attribute__((constructor)) void init() { // lefut a program futás elején automatikusan
    invi[1] = w[N / 2] = 1;
    constexpr long long g = mod_pow(mod_primitive_root(), mod / N);
    // const double PI = acos(-1); double ang = 2 * PI / n * (invert ? -1 : 1);
    for (int i = N / 2 + 1; i < N; ++i) w[i] = w[i - 1] * g % mod; // complex<double>(cos(ang * (i - N / 2)), sin(ang * (i - N / 2)));
    for (int i = N / 2 - 1; i > 0; --i) w[i] = w[i << 1];
    for (int i = 2; i <= N; i++) invi[i] = invi[mod % i] * (mod - mod / i) % mod; // csak a log-hoz kell
}
void dft(long long f[], int n) { // n ketto hatvány
    assert(__builtin_popcount(n) == 1);
    for (int k = n / 2; k; k /= 2)
        for (int i = 0; i < n; i += k + k)
            for (int j = 0; j < k; ++j) {
                long long x = f[i + j]; long long y = f[i + j + k]; f[i + j] = (x + y) % mod; f[i + j + k] = (mod + x - y) * w[k + j] % mod;
            }
}
void ift(long long f[], int n) { // n ketto hatvány
    assert(__builtin_popcount(n) == 1);
    for (int k = 1; k < n; k *= 2)
        for (int i = 0; i < n; i += k + k)
            for (int j = 0; j < k; ++j) {
                long long x = f[i + j]; long long y = f[i + j + k] * w[k + j] % mod; f[i + j] = (x + y) % mod; f[i + j + k] = (mod + x - y) % mod;
            }
    long long inv = mod - (mod - 1) / n;
    std::reverse(f + 1, f + n);
    for (int i = 0; i < n; ++i) f[i] = f[i] * inv % mod;
}

void dft(vector<long long> &a, int n){
    assert(__builtin_popcount(n) == 1);
    a.resize(n);
    dft(a.data(), n);
}

void ift(vector<long long> &a, int n){
    assert(__builtin_popcount(n) == 1);
    a.resize(n);
    ift(a.data(), n);
}

void add(vector<long long> &a, const vector<long long> &b){ if(a.size() < b.size()) a.resize(b.size()); for(int i = 0; i < (int)b.size(); i++) a[i] = (a[i] + b[i]) % mod; }
void sub(vector<long long> &a, const vector<long long> &b){ if(a.size() < b.size()) a.resize(b.size()); for(int i = 0; i < (int)b.size(); i++) a[i] = (a[i] - b[i] + mod) % mod; }
void mul(vector<long long> &a, const vector<long long> &b){ int n = min(a.size(), b.size()); a.resize(n); for(int i = 0; i < n; i++) a[i] = a[i] * b[i] % mod; }
void mul(vector<long long> &a, long long t){ for(int i = 0; i < (int)a.size(); i++) a[i] = a[i] * t % mod; }
void derivate(vector<long long> &a) { for(int i = 0; i < (int)a.size()-1; i++) a[i] = a[i + 1] * (i + 1) % mod; a.pop_back(); }
void integrate(vector<long long> &a) { a.push_back(0); for(int i = (int)a.size() - 1; i > 0; i--) a[i] = a[i - 1] * invi[i] % mod; a[0] = 0; } // invi[] méretére figyelni kell
void del(vector<long long> &a, int s) { a.erase(a.begin(), a.begin() + min(s, (int)a.size())); }
void ins(vector<long long> &a, int s) { a.insert(a.begin(), s, 0); }

vector<long long> conv(vector<long long> a, vector<long long> b){
    int sz = (int)a.size() + (int)b.size() - 1, n = __bit_ceil(max(sz, 0));
    a.resize(n), b.resize(n);
    dft(a, n); dft(b, n); mul(a, b); ift(a, n);
    return vector<long long>(a.begin(), a.begin() + sz);
}

vector<long long> inv(const vector<long long>& a) {
    if(a[0] == 0) return vector<long long>{};
    int m = a.size();
    vector<long long> res = {mod_pow(a[0], mod - 2)};
    for(int k = 1; k < m; k *= 2){
        int n = k * 2;
        res.resize(n);
        vector<long long> p = res; 
        vector<long long> q(a.begin(), a.begin() + min((int)a.size(), n));
        dft(p, n);
        dft(q, n); 
        mul(q, p); ift(q, n); del(q, k); dft(q, n); mul(p, q); ift(p, n); ins(p, k); p.resize(n); sub(res, p);
    }
    return vector<long long>(res.begin(), res.begin() + m);
}

/*
Polinom művelet Hensel Lemmával: q_0 úgy kell választani, hogy konstans jó legyen | q_{k+1} = q_{k} - F(q_{k}) / F'(q_{k}) | q_k a x^2^k-ra jó
pow: elején lévő 0-at le kell venni utána be kell szorozni c-vel, hogy p[0] = 1 legyen, utána log() -> *k -> exp() -> * c^{-k} -> + 0-k az elejére
sqrt: ellenőrizni kell, hogy p[0]-nak van-e gyöke: 0-t leszedni: pow()-al 1/2-re elmelni -> c-vel felszorozni -> vissza rakni a 0-t (csak akkor jó ha páros sok 0 van az elején)
*/

vector<long long> log(vector<long long> a){ // res[0] = 0
    int n = __bit_ceil(a.size() * 2 - 1), sz(a.size());
    vector<long long> a_inv = inv(a); a_inv.resize(n);
    derivate(a); dft(a, n); dft(a_inv, n); mul(a, a_inv); ift(a, n); integrate(a); a.resize(sz);
    return a;
}

vector<long long> exp(const vector<long long>& a) { // p[0] == 0, különben nem valid az eremény
    if (a[0] != 0) return {};
    int m = a.size();
    vector<long long> e = {1};
    for (int k = 1; k < m; k *= 2) {
        int n = k * 2;
        vector<long long> elog(e); elog.resize(n); elog = log(elog);
        dft(e, n*2);
        vector<long long> p(a.begin(), a.begin() + min((int)a.size(), n));
        sub(p, elog); dft(p, n*2); mul(p, e);
        add(e, p); ift(e, n*2); e.resize(n);
    }
    e.resize(m);
    return e;
}

vector<long long> div(vector<long long> a, vector<long long> b) { // a / b
    while(!a.size() && a.back() == 0) a.pop_back();
    while(!b.size() && b.back() == 0) b.pop_back();
    int m = (int)a.size() - (int)b.size() + 1; // kell az int!!!
    if(a.empty() || b.empty() || a.size() < b.size()) return b.empty() ? vector<long long>{} : vector<long long>{0};
    reverse(a.begin(), a.end()); reverse(b.begin(), b.end());
    b.resize(a.size());
    b = inv(b);
    a = conv(a, b);
    a.resize(m);
    reverse(a.begin(), a.end());
    return a;
}

vector<long long> rem(vector<long long> a, vector<long long> b) { sub(a, conv(div(a, b), b)); return a; }
